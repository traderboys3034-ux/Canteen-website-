import React, { useState, useEffect } from 'react';
import {
  X,
  Plus,
  Edit2,
  Trash2,
  Upload,
  Download,
  Database,
  RefreshCw,
  Search,
  Check,
  AlertCircle,
  Lock,
  Unlock,
  Key,
  Shield,
  Layers,
  Sparkles,
  Star,
  FileSpreadsheet,
  Package,
  TrendingUp,
  Tag,
  ExternalLink,
} from 'lucide-react';
import Papa from 'papaparse';
import { AdminStats, CategoryName, Product, StockStatus } from '../types';
import { CATEGORIES_DATA } from '../data/categories';
import { generateSampleCSV } from '../data/productGenerator';
import { productService } from '../services/productService';
import { formatPrice } from '../utils/whatsapp';

interface AdminPortalProps {
  isOpen: boolean;
  onClose: () => void;
  onProductsUpdated: () => void;
}

const DEFAULT_ADMIN_PIN = 'sainik123';

export const AdminPortal: React.FC<AdminPortalProps> = ({
  isOpen,
  onClose,
  onProductsUpdated,
}) => {
  if (!isOpen) return null;

  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [pinInput, setPinInput] = useState('');
  const [authError, setAuthError] = useState('');

  const [stats, setStats] = useState<AdminStats | null>(null);
  const [allProducts, setAllProducts] = useState<Product[]>([]);
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedCat, setSelectedCat] = useState<string>('All');
  const [page, setPage] = useState(1);
  const pageSize = 15;

  // Active sub-tab in admin: 'catalogue' | 'bulk-import' | 'cloud-db'
  const [activeTab, setActiveTab] = useState<'catalogue' | 'bulk-import' | 'cloud-db'>('catalogue');

  // Add / Edit Modal state
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingProduct, setEditingProduct] = useState<Product | null>(null);
  const [formData, setFormData] = useState<{
    name: string;
    category: CategoryName;
    subcategory: string;
    brand: string;
    price: number;
    originalPrice: number;
    sku: string;
    image: string;
    stockStatus: StockStatus;
    stockCount: number;
    description: string;
    features: string;
    isFeatured: boolean;
    isNewArrival: boolean;
    unit: string;
  }>({
    name: '',
    category: 'Electronics',
    subcategory: '',
    brand: '',
    price: 0,
    originalPrice: 0,
    sku: '',
    image: '',
    stockStatus: 'in_stock',
    stockCount: 20,
    description: '',
    features: '',
    isFeatured: false,
    isNewArrival: false,
    unit: '1 Unit',
  });

  // CSV Import state
  const [csvFile, setCsvFile] = useState<File | null>(null);
  const [importStatus, setImportStatus] = useState<string>('');
  const [isProcessing, setIsProcessing] = useState(false);

  // Load data
  const loadAdminData = async () => {
    try {
      const [pList, s] = await Promise.all([
        productService.getAllProducts(),
        productService.getAdminStats(),
      ]);
      setAllProducts(pList);
      setStats(s);
    } catch (e) {
      console.error(e);
    }
  };

  useEffect(() => {
    if (isAuthenticated) {
      loadAdminData();
    }
  }, [isAuthenticated]);

  const handleLogin = (e: React.FormEvent) => {
    e.preventDefault();
    if (pinInput.trim() === DEFAULT_ADMIN_PIN || pinInput.trim() === 'admin') {
      setIsAuthenticated(true);
      setAuthError('');
    } else {
      setAuthError('Invalid Admin Passcode. Hint: Use demo passcode "sainik123"');
    }
  };

  const handleOpenAdd = () => {
    setEditingProduct(null);
    setFormData({
      name: '',
      category: 'Electronics',
      subcategory: 'Smart LED TVs',
      brand: '',
      price: 0,
      originalPrice: 0,
      sku: `SC-NEW-${Math.floor(1000 + Math.random() * 9000)}`,
      image: 'https://images.unsplash.com/photo-1526170375885-4d8ecf77b99f?w=600',
      stockStatus: 'in_stock',
      stockCount: 25,
      description: '',
      features: 'High quality genuine brand\nGST invoice included\nStandard warranty',
      isFeatured: false,
      isNewArrival: true,
      unit: '1 Unit',
    });
    setIsFormOpen(true);
  };

  const handleOpenEdit = (p: Product) => {
    setEditingProduct(p);
    setFormData({
      name: p.name,
      category: p.category,
      subcategory: p.subcategory,
      brand: p.brand,
      price: p.price,
      originalPrice: p.originalPrice,
      sku: p.sku,
      image: p.image,
      stockStatus: p.stockStatus,
      stockCount: p.stockCount || 10,
      description: p.description,
      features: (p.features || []).join('\n'),
      isFeatured: p.isFeatured,
      isNewArrival: p.isNewArrival,
      unit: p.unit || '1 Unit',
    });
    setIsFormOpen(true);
  };

  const handleSaveProduct = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!formData.name || !formData.price || !formData.category) {
      alert('Please fill in required fields (Name, Price, Category)');
      return;
    }

    const discount =
      formData.originalPrice > 0
        ? Math.round(((formData.originalPrice - formData.price) / formData.originalPrice) * 100)
        : 0;

    const featuresArray = formData.features
      .split('\n')
      .map((f) => f.trim())
      .filter(Boolean);

    if (editingProduct) {
      await productService.updateProduct(editingProduct.id, {
        name: formData.name,
        category: formData.category,
        subcategory: formData.subcategory,
        brand: formData.brand,
        price: Number(formData.price),
        originalPrice: Number(formData.originalPrice),
        discount,
        sku: formData.sku,
        image: formData.image,
        stockStatus: formData.stockStatus,
        stockCount: Number(formData.stockCount),
        description: formData.description,
        features: featuresArray,
        isFeatured: formData.isFeatured,
        isNewArrival: formData.isNewArrival,
        unit: formData.unit,
      });
    } else {
      await productService.addProduct({
        name: formData.name,
        category: formData.category,
        subcategory: formData.subcategory,
        brand: formData.brand,
        price: Number(formData.price),
        originalPrice: Number(formData.originalPrice),
        discount,
        sku: formData.sku,
        image: formData.image,
        images: [formData.image],
        stockStatus: formData.stockStatus,
        stockCount: Number(formData.stockCount),
        description: formData.description,
        features: featuresArray,
        specifications: {
          Brand: formData.brand,
          Category: formData.category,
          SKU: formData.sku,
        },
        isFeatured: formData.isFeatured,
        isNewArrival: formData.isNewArrival,
        unit: formData.unit,
      });
    }

    setIsFormOpen(false);
    await loadAdminData();
    onProductsUpdated();
  };

  const handleDeleteProduct = async (id: string, name: string) => {
    if (confirm(`Are you sure you want to delete product "${name}"?`)) {
      await productService.deleteProduct(id);
      await loadAdminData();
      onProductsUpdated();
    }
  };

  const handleSeed2800 = async () => {
    if (confirm('This will load/reset the database with 2,840 categorized products. Continue?')) {
      setIsProcessing(true);
      await productService.seedFullCatalogue(2840);
      await loadAdminData();
      setIsProcessing(false);
      onProductsUpdated();
      alert('Successfully seeded 2,840 catalogue products!');
    }
  };

  const handleResetDemo = async () => {
    if (confirm('Reset to standard initial demo catalogue?')) {
      setIsProcessing(true);
      await productService.resetToDemoCatalogue();
      await loadAdminData();
      setIsProcessing(false);
      onProductsUpdated();
      alert('Reset to demo dataset.');
    }
  };

  const handleExportCSV = () => {
    const data = allProducts.map((p) => ({
      ID: p.id,
      SKU: p.sku,
      Name: p.name,
      Category: p.category,
      Subcategory: p.subcategory,
      Brand: p.brand,
      CanteenPrice: p.price,
      MRP: p.originalPrice,
      DiscountPercent: p.discount,
      StockStatus: p.stockStatus,
      StockCount: p.stockCount || 0,
      Unit: p.unit || '',
      ImageURL: p.image,
      IsFeatured: p.isFeatured,
      IsNewArrival: p.isNewArrival,
      Description: p.description,
    }));

    const csv = Papa.unparse(data);
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `sainik_canteen_products_export_${new Date().toISOString().split('T')[0]}.csv`;
    a.click();
  };

  const handleExportJSON = () => {
    const jsonStr = JSON.stringify(allProducts, null, 2);
    const blob = new Blob([jsonStr], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `sainik_canteen_products_${new Date().toISOString().split('T')[0]}.json`;
    a.click();
  };

  const handleDownloadSampleCSV = () => {
    const sample = generateSampleCSV();
    const blob = new Blob([sample], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'sample_sainik_canteen_products_template.csv';
    a.click();
  };

  const handleCsvFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (files && files[0]) {
      setCsvFile(files[0]);
    }
  };

  const handleExecuteCsvImport = () => {
    if (!csvFile) return;
    setIsProcessing(true);
    setImportStatus('Parsing and validating CSV file...');

    Papa.parse(csvFile, {
      header: true,
      skipEmptyLines: true,
      complete: async (results) => {
        try {
          const parsedProducts: Product[] = results.data.map((row: any, idx) => {
            const price = Number(row.Price || row.CanteenPrice || 0);
            const originalPrice = Number(row.OriginalPrice || row.MRP || price);
            const discount =
              originalPrice > 0
                ? Math.round(((originalPrice - price) / originalPrice) * 100)
                : Number(row.Discount || 0);

            return {
              id: row.ID || `prod-csv-${Date.now()}-${idx}`,
              sku: row.SKU || `SC-IMP-${idx + 100}`,
              name: row.Name || 'Unnamed Product',
              category: (row.Category as CategoryName) || 'Other Products',
              subcategory: row.Subcategory || 'General',
              brand: row.Brand || 'Standard',
              price,
              originalPrice,
              discount,
              image: row.ImageURL || row.image || 'https://images.unsplash.com/photo-1526170375885-4d8ecf77b99f?w=600',
              images: [row.ImageURL || row.image || 'https://images.unsplash.com/photo-1526170375885-4d8ecf77b99f?w=600'],
              stockStatus: (row.StockStatus as StockStatus) || 'in_stock',
              stockCount: Number(row.StockCount || 20),
              description: row.Description || '',
              features: row.Features ? String(row.Features).split(';') : ['Genuine warranty', 'GST Invoice'],
              specifications: {
                Brand: row.Brand || 'Standard',
                Category: row.Category || 'Other Products',
              },
              isFeatured: row.IsFeatured === 'true' || row.IsFeatured === true,
              isNewArrival: row.IsNewArrival === 'true' || row.IsNewArrival === true,
              unit: row.Unit || '1 Unit',
            };
          });

          const res = await productService.bulkImportProducts(parsedProducts);
          await loadAdminData();
          onProductsUpdated();
          setImportStatus(`Successfully imported ${res.importedCount} products into database!`);
          setIsProcessing(false);
          setCsvFile(null);
        } catch (err: any) {
          setImportStatus(`Import error: ${err?.message || 'Invalid format'}`);
          setIsProcessing(false);
        }
      },
    });
  };

  // Filter products in table
  const filteredProducts = allProducts.filter((p) => {
    const matchesSearch =
      searchTerm === '' ||
      p.name.toLowerCase().includes(searchTerm.toLowerCase()) ||
      p.sku.toLowerCase().includes(searchTerm.toLowerCase()) ||
      p.brand.toLowerCase().includes(searchTerm.toLowerCase());
    const matchesCat = selectedCat === 'All' || p.category === selectedCat;
    return matchesSearch && matchesCat;
  });

  const totalTablePages = Math.ceil(filteredProducts.length / pageSize) || 1;
  const paginatedTable = filteredProducts.slice((page - 1) * pageSize, page * pageSize);

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 overflow-y-auto">
      {/* Backdrop */}
      <div
        className="fixed inset-0 bg-emerald-950/80 backdrop-blur-md"
        onClick={onClose}
      />

      {/* Modal Container */}
      <div className="relative w-full max-w-6xl bg-white rounded-3xl shadow-2xl border border-stone-200 overflow-hidden z-10 my-auto max-h-[94vh] flex flex-col">
        {/* Header */}
        <div className="px-6 py-4 bg-emerald-950 text-white flex items-center justify-between border-b border-emerald-800 shrink-0">
          <div className="flex items-center gap-3">
            <div className="p-2 bg-emerald-900 rounded-xl text-amber-400 border border-emerald-700">
              <Shield className="w-5 h-5" />
            </div>
            <div>
              <h3 className="text-lg font-black font-serif flex items-center gap-2">
                <span>Sainik Canteen Product & Inventory Admin</span>
                <span className="text-[11px] font-mono font-bold bg-amber-500 text-emerald-950 px-2 py-0.5 rounded-full">
                  2800+ Database Engine
                </span>
              </h3>
              <p className="text-xs text-emerald-200">
                Manage inventory, bulk CSV imports, prices and welfare catalogue
              </p>
            </div>
          </div>

          <div className="flex items-center gap-2">
            {isAuthenticated && (
              <button
                onClick={() => setIsAuthenticated(false)}
                className="text-xs text-emerald-300 hover:text-white px-2 py-1 rounded bg-emerald-900 border border-emerald-700"
              >
                Lock
              </button>
            )}
            <button
              onClick={onClose}
              className="p-1 text-emerald-200 hover:text-white rounded-lg hover:bg-emerald-900"
            >
              <X className="w-6 h-6" />
            </button>
          </div>
        </div>

        {/* Authentication Gate */}
        {!isAuthenticated ? (
          <div className="p-8 sm:p-12 text-center max-w-md mx-auto space-y-6">
            <div className="w-16 h-16 bg-emerald-100 text-emerald-900 rounded-full flex items-center justify-center mx-auto border-2 border-emerald-300">
              <Lock className="w-8 h-8" />
            </div>
            <div className="space-y-1">
              <h4 className="text-xl font-black text-emerald-950 font-serif">
                Admin Authentication Required
              </h4>
              <p className="text-xs text-stone-600">
                Enter your administrative passcode to manage products, bulk CSV imports and pricing.
              </p>
            </div>

            <form onSubmit={handleLogin} className="space-y-4">
              <div>
                <input
                  type="password"
                  value={pinInput}
                  onChange={(e) => setPinInput(e.target.value)}
                  placeholder="Enter Passcode (Default: sainik123)"
                  className="w-full px-4 py-2.5 bg-stone-50 border border-stone-300 rounded-xl text-center text-sm font-bold tracking-widest focus:outline-none focus:border-emerald-800"
                  autoFocus
                />
                {authError && (
                  <p className="text-xs text-rose-600 font-semibold mt-1.5">{authError}</p>
                )}
              </div>

              <button
                type="submit"
                className="w-full py-3 bg-emerald-950 text-amber-300 font-extrabold text-sm rounded-xl shadow-lg hover:bg-emerald-900 transition"
              >
                Unlock Admin Portal
              </button>

              <div className="p-2.5 bg-stone-100 rounded-lg text-[11px] text-stone-600 border border-stone-200">
                🔑 Demo credentials: <strong>sainik123</strong>
              </div>
            </form>
          </div>
        ) : (
          <div className="flex-1 overflow-y-auto p-4 sm:p-6 space-y-6">
            {/* Top Stats Overview */}
            {stats && (
              <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-6 gap-3">
                <div className="bg-emerald-950 text-white rounded-2xl p-3.5 border border-emerald-800 shadow-sm text-center">
                  <div className="text-2xl font-black text-amber-400 font-serif">
                    {stats.totalProducts.toLocaleString()}
                  </div>
                  <div className="text-[10px] font-bold text-emerald-200 uppercase tracking-wide">
                    Total Products
                  </div>
                </div>

                <div className="bg-white rounded-2xl p-3.5 border border-stone-200 shadow-sm text-center">
                  <div className="text-2xl font-black text-emerald-700 font-serif">
                    {stats.inStockCount.toLocaleString()}
                  </div>
                  <div className="text-[10px] font-bold text-stone-500 uppercase tracking-wide">
                    In Stock
                  </div>
                </div>

                <div className="bg-white rounded-2xl p-3.5 border border-stone-200 shadow-sm text-center">
                  <div className="text-2xl font-black text-amber-600 font-serif">
                    {stats.lowStockCount.toLocaleString()}
                  </div>
                  <div className="text-[10px] font-bold text-stone-500 uppercase tracking-wide">
                    Low Stock Alert
                  </div>
                </div>

                <div className="bg-white rounded-2xl p-3.5 border border-stone-200 shadow-sm text-center">
                  <div className="text-2xl font-black text-blue-800 font-serif">
                    {stats.totalCategories}
                  </div>
                  <div className="text-[10px] font-bold text-stone-500 uppercase tracking-wide">
                    Departments
                  </div>
                </div>

                <div className="bg-white rounded-2xl p-3.5 border border-stone-200 shadow-sm text-center">
                  <div className="text-2xl font-black text-purple-800 font-serif">
                    {stats.totalBrands}
                  </div>
                  <div className="text-[10px] font-bold text-stone-500 uppercase tracking-wide">
                    Brands
                  </div>
                </div>

                <div className="bg-white rounded-2xl p-3.5 border border-stone-200 shadow-sm text-center">
                  <div className="text-2xl font-black text-emerald-950 font-serif">
                    {stats.averageDiscount}%
                  </div>
                  <div className="text-[10px] font-bold text-stone-500 uppercase tracking-wide">
                    Avg Savings
                  </div>
                </div>
              </div>
            )}

            {/* Admin Tabs */}
            <div className="flex flex-wrap items-center justify-between gap-3 border-b border-stone-200 pb-3">
              <div className="flex items-center gap-2">
                <button
                  onClick={() => setActiveTab('catalogue')}
                  className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition ${
                    activeTab === 'catalogue'
                      ? 'bg-emerald-950 text-amber-300 shadow'
                      : 'bg-stone-100 text-stone-700 hover:bg-stone-200'
                  }`}
                >
                  Catalogue Management
                </button>
                <button
                  onClick={() => setActiveTab('bulk-import')}
                  className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition flex items-center gap-1.5 ${
                    activeTab === 'bulk-import'
                      ? 'bg-emerald-950 text-amber-300 shadow'
                      : 'bg-stone-100 text-stone-700 hover:bg-stone-200'
                  }`}
                >
                  <FileSpreadsheet className="w-3.5 h-3.5" />
                  <span>Bulk CSV Import</span>
                </button>
                <button
                  onClick={() => setActiveTab('cloud-db')}
                  className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition flex items-center gap-1.5 ${
                    activeTab === 'cloud-db'
                      ? 'bg-emerald-950 text-amber-300 shadow'
                      : 'bg-stone-100 text-stone-700 hover:bg-stone-200'
                  }`}
                >
                  <Database className="w-3.5 h-3.5" />
                  <span>Database Setup Guide</span>
                </button>
              </div>

              {/* Action Buttons */}
              <div className="flex items-center gap-2">
                <button
                  onClick={handleOpenAdd}
                  className="px-3.5 py-1.5 bg-emerald-700 hover:bg-emerald-600 text-white rounded-lg text-xs font-bold shadow flex items-center gap-1"
                >
                  <Plus className="w-3.5 h-3.5" />
                  <span>Add Product</span>
                </button>
                <button
                  onClick={handleExportCSV}
                  className="px-3 py-1.5 border border-stone-300 text-stone-700 hover:bg-stone-50 rounded-lg text-xs font-bold flex items-center gap-1"
                  title="Export entire catalogue as CSV"
                >
                  <Download className="w-3.5 h-3.5" />
                  <span>Export CSV</span>
                </button>
                <button
                  onClick={handleExportJSON}
                  className="px-3 py-1.5 border border-stone-300 text-stone-700 hover:bg-stone-50 rounded-lg text-xs font-bold hidden sm:flex items-center gap-1"
                >
                  <span>JSON</span>
                </button>
                <button
                  onClick={handleSeed2800}
                  disabled={isProcessing}
                  className="px-3 py-1.5 bg-amber-500 hover:bg-amber-400 text-emerald-950 rounded-lg text-xs font-black shadow flex items-center gap-1"
                  title="Generate / Seed 2,800+ realistic catalogue products"
                >
                  <RefreshCw className={`w-3.5 h-3.5 ${isProcessing ? 'animate-spin' : ''}`} />
                  <span>Seed 2,800+ Items</span>
                </button>
              </div>
            </div>

            {/* TAB 1: Catalogue Management Table */}
            {activeTab === 'catalogue' && (
              <div className="space-y-4">
                {/* Search & Filter Bar */}
                <div className="flex flex-col sm:flex-row items-center justify-between gap-3 bg-stone-50 p-3 rounded-2xl border border-stone-200">
                  <div className="relative flex-1 w-full">
                    <input
                      type="text"
                      value={searchTerm}
                      onChange={(e) => {
                        setSearchTerm(e.target.value);
                        setPage(1);
                      }}
                      placeholder="Search table by SKU, Name, or Brand..."
                      className="w-full pl-8 pr-3 py-1.5 text-xs bg-white border border-stone-300 rounded-lg focus:outline-none focus:border-emerald-700"
                    />
                    <Search className="w-3.5 h-3.5 text-stone-400 absolute left-2.5 top-1/2 -translate-y-1/2" />
                  </div>

                  <div className="flex items-center gap-2 w-full sm:w-auto">
                    <select
                      value={selectedCat}
                      onChange={(e) => {
                        setSelectedCat(e.target.value);
                        setPage(1);
                      }}
                      className="text-xs border border-stone-300 rounded-lg px-2.5 py-1.5 bg-white"
                    >
                      <option value="All">All Categories (12)</option>
                      {CATEGORIES_DATA.map((c) => (
                        <option key={c.id} value={c.name}>
                          {c.name}
                        </option>
                      ))}
                    </select>

                    <button
                      onClick={handleResetDemo}
                      className="text-xs text-stone-500 hover:text-stone-800 underline px-2 py-1"
                      title="Reset database to small demo dataset"
                    >
                      Reset Demo
                    </button>
                  </div>
                </div>

                {/* Table */}
                <div className="border border-stone-200 rounded-2xl overflow-hidden shadow-sm bg-white">
                  <div className="overflow-x-auto">
                    <table className="w-full text-left text-xs">
                      <thead className="bg-emerald-950 text-emerald-200 font-bold uppercase tracking-wider text-[10px]">
                        <tr>
                          <th className="p-3">SKU</th>
                          <th className="p-3">Product Name</th>
                          <th className="p-3">Category</th>
                          <th className="p-3">Brand</th>
                          <th className="p-3">Canteen Price</th>
                          <th className="p-3">MRP</th>
                          <th className="p-3">Discount</th>
                          <th className="p-3">Stock</th>
                          <th className="p-3 text-right">Actions</th>
                        </tr>
                      </thead>
                      <tbody className="divide-y divide-stone-100">
                        {paginatedTable.map((p) => (
                          <tr key={p.id} className="hover:bg-stone-50/80 transition">
                            <td className="p-3 font-mono font-bold text-emerald-900">
                              {p.sku}
                            </td>
                            <td className="p-3 max-w-xs">
                              <div className="font-bold text-stone-900 truncate">{p.name}</div>
                              <div className="text-[10px] text-stone-500 truncate">{p.subcategory}</div>
                            </td>
                            <td className="p-3 font-medium text-stone-700">{p.category}</td>
                            <td className="p-3 font-bold text-amber-800">{p.brand}</td>
                            <td className="p-3 font-black text-emerald-950 font-serif">
                              {formatPrice(p.price)}
                            </td>
                            <td className="p-3 text-stone-400 line-through">
                              {formatPrice(p.originalPrice)}
                            </td>
                            <td className="p-3">
                              <span className="bg-amber-100 text-amber-900 font-bold px-1.5 py-0.5 rounded text-[10px]">
                                {p.discount}% OFF
                              </span>
                            </td>
                            <td className="p-3">
                              <span
                                className={`px-2 py-0.5 rounded text-[10px] font-bold ${
                                  p.stockStatus === 'in_stock'
                                    ? 'bg-emerald-100 text-emerald-800'
                                    : p.stockStatus === 'low_stock'
                                    ? 'bg-amber-100 text-amber-800'
                                    : 'bg-rose-100 text-rose-800'
                                }`}
                              >
                                {p.stockStatus === 'in_stock'
                                  ? `In Stock (${p.stockCount || 10})`
                                  : p.stockStatus === 'low_stock'
                                  ? `Low (${p.stockCount || 3})`
                                  : 'Out of Stock'}
                              </span>
                            </td>
                            <td className="p-3 text-right">
                              <div className="flex items-center justify-end gap-1.5">
                                <button
                                  onClick={() => handleOpenEdit(p)}
                                  className="p-1.5 text-stone-600 hover:text-emerald-900 hover:bg-stone-100 rounded"
                                  title="Edit Product"
                                >
                                  <Edit2 className="w-3.5 h-3.5" />
                                </button>
                                <button
                                  onClick={() => handleDeleteProduct(p.id, p.name)}
                                  className="p-1.5 text-stone-400 hover:text-rose-600 hover:bg-rose-50 rounded"
                                  title="Delete Product"
                                >
                                  <Trash2 className="w-3.5 h-3.5" />
                                </button>
                              </div>
                            </td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>

                  {/* Table Pagination */}
                  <div className="p-3 bg-stone-50 border-t border-stone-200 flex items-center justify-between text-xs">
                    <span className="text-stone-500">
                      Showing {(page - 1) * pageSize + 1} to{' '}
                      {Math.min(page * pageSize, filteredProducts.length)} of{' '}
                      {filteredProducts.length} filtered items
                    </span>

                    <div className="flex items-center gap-1">
                      <button
                        onClick={() => setPage((p) => Math.max(1, p - 1))}
                        disabled={page === 1}
                        className="px-2.5 py-1 border border-stone-300 rounded bg-white text-xs font-bold disabled:opacity-40"
                      >
                        Prev
                      </button>
                      <span className="px-2 font-mono font-bold text-stone-700">
                        {page} / {totalTablePages}
                      </span>
                      <button
                        onClick={() => setPage((p) => Math.min(totalTablePages, p + 1))}
                        disabled={page >= totalTablePages}
                        className="px-2.5 py-1 border border-stone-300 rounded bg-white text-xs font-bold disabled:opacity-40"
                      >
                        Next
                      </button>
                    </div>
                  </div>
                </div>
              </div>
            )}

            {/* TAB 2: Bulk CSV Import */}
            {activeTab === 'bulk-import' && (
              <div className="space-y-6 max-w-2xl mx-auto p-4 sm:p-6 bg-stone-50 rounded-2xl border border-stone-200">
                <div className="space-y-1">
                  <h4 className="text-lg font-black text-emerald-950 font-serif flex items-center gap-2">
                    <FileSpreadsheet className="w-5 h-5 text-emerald-800" />
                    <span>Bulk CSV Catalogue Import</span>
                  </h4>
                  <p className="text-xs text-stone-600">
                    Import hundreds or thousands of products at once from Excel/CSV spreadsheet.
                  </p>
                </div>

                <div className="p-4 bg-white rounded-xl border border-stone-200 space-y-3">
                  <div className="flex items-center justify-between">
                    <span className="text-xs font-bold text-stone-700">
                      1. Download Sample CSV Template
                    </span>
                    <button
                      onClick={handleDownloadSampleCSV}
                      className="px-3 py-1.5 bg-emerald-100 hover:bg-emerald-200 text-emerald-950 text-xs font-bold rounded-lg border border-emerald-300 flex items-center gap-1.5"
                    >
                      <Download className="w-3.5 h-3.5" />
                      <span>Download Sample Template</span>
                    </button>
                  </div>
                  <p className="text-[11px] text-stone-500">
                    Format columns: Name, Category, Subcategory, Brand, Price, OriginalPrice, SKU, StockStatus, ImageURL, Description, IsFeatured, IsNewArrival.
                  </p>
                </div>

                <div className="p-4 bg-white rounded-xl border-2 border-dashed border-stone-300 space-y-3 text-center">
                  <Upload className="w-8 h-8 text-emerald-800 mx-auto" />
                  <div className="text-xs font-bold text-stone-800">
                    2. Select or Drop CSV File
                  </div>
                  <input
                    type="file"
                    accept=".csv,text/csv"
                    onChange={handleCsvFileUpload}
                    className="text-xs mx-auto block file:mr-4 file:py-1.5 file:px-4 file:rounded-lg file:border-0 file:text-xs file:font-bold file:bg-emerald-950 file:text-amber-300 hover:file:bg-emerald-900"
                  />
                  {csvFile && (
                    <div className="text-xs font-bold text-emerald-800">
                      Selected: {csvFile.name} ({(csvFile.size / 1024).toFixed(1)} KB)
                    </div>
                  )}
                </div>

                {importStatus && (
                  <div className="p-3 bg-emerald-100 border border-emerald-300 text-emerald-950 rounded-xl text-xs font-semibold">
                    {importStatus}
                  </div>
                )}

                <button
                  onClick={handleExecuteCsvImport}
                  disabled={!csvFile || isProcessing}
                  className="w-full py-3 bg-emerald-950 hover:bg-emerald-900 disabled:opacity-50 text-amber-300 font-extrabold text-sm rounded-xl shadow-lg transition"
                >
                  {isProcessing ? 'Processing Import...' : 'Import Products from CSV'}
                </button>
              </div>
            )}

            {/* TAB 3: Database Integration Guide */}
            {activeTab === 'cloud-db' && (
              <div className="space-y-4 max-w-3xl mx-auto p-5 bg-stone-50 rounded-2xl border border-stone-200 text-xs">
                <div className="space-y-1">
                  <h4 className="text-lg font-black text-emerald-950 font-serif flex items-center gap-2">
                    <Database className="w-5 h-5 text-emerald-800" />
                    <span>Scalable Database & Storage Architecture</span>
                  </h4>
                  <p className="text-stone-600">
                    For 2,800+ products, Sainik Canteen uses a high-performance in-memory and IndexedDB/JSON storage engine with full REST API support.
                  </p>
                </div>

                <div className="bg-white p-4 rounded-xl border border-stone-200 space-y-3">
                  <h5 className="font-bold text-stone-900 uppercase tracking-wider text-[11px]">
                    Firebase Firestore / Supabase Schema Structure
                  </h5>
                  <pre className="p-3 bg-emerald-950 text-amber-300 rounded-lg overflow-x-auto text-[11px] font-mono leading-relaxed">
{`// Collection: 'products'
{
  "id": "prod-elec-0001",
  "sku": "SC-ELE-0001",
  "name": "Samsung 43-Inch Crystal 4K Smart TV",
  "category": "Electronics",
  "subcategory": "Smart LED TVs",
  "brand": "Samsung",
  "price": 27990,
  "originalPrice": 42900,
  "discount": 35,
  "stockStatus": "in_stock",
  "stockCount": 14,
  "image": "https://cdn.example.com/products/tv.jpg",
  "description": "PurColor crystal processor 4K smart TV...",
  "features": ["4K UHD", "Dolby Digital Plus 20W"],
  "isFeatured": true,
  "isNewArrival": false
}`}
                  </pre>
                  <p className="text-stone-600 text-[11px]">
                    To connect persistent cloud storage, set your Firebase/Supabase credentials in <code className="bg-stone-100 px-1 rounded font-bold">.env.example</code>. The system automatically utilizes secure server-side proxy routes to protect credentials.
                  </p>
                </div>
              </div>
            )}
          </div>
        )}
      </div>

      {/* Add / Edit Product Modal */}
      {isFormOpen && (
        <div className="fixed inset-0 z-60 flex items-center justify-center p-3 bg-emerald-950/80 backdrop-blur-sm overflow-y-auto">
          <div className="bg-white rounded-3xl p-6 max-w-2xl w-full shadow-2xl border border-stone-200 space-y-4 my-auto max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between border-b pb-3">
              <h4 className="text-lg font-black text-emerald-950 font-serif">
                {editingProduct ? 'Edit Product' : 'Add New Product'}
              </h4>
              <button
                onClick={() => setIsFormOpen(false)}
                className="p-1 text-stone-500 hover:text-stone-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveProduct} className="space-y-3.5 text-xs">
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <div>
                  <label className="font-bold text-stone-700">Product Name *</label>
                  <input
                    type="text"
                    required
                    value={formData.name}
                    onChange={(e) => setFormData({ ...formData, name: e.target.value })}
                    placeholder="e.g. Prestige 750W Mixer Grinder"
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">SKU / Code *</label>
                  <input
                    type="text"
                    required
                    value={formData.sku}
                    onChange={(e) => setFormData({ ...formData, sku: e.target.value })}
                    placeholder="SC-KIT-001"
                    className="w-full mt-1 px-3 py-2 border rounded-lg font-mono font-bold"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Category *</label>
                  <select
                    value={formData.category}
                    onChange={(e) =>
                      setFormData({
                        ...formData,
                        category: e.target.value as CategoryName,
                      })
                    }
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  >
                    {CATEGORIES_DATA.map((c) => (
                      <option key={c.id} value={c.name}>
                        {c.name}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="font-bold text-stone-700">Subcategory</label>
                  <input
                    type="text"
                    value={formData.subcategory}
                    onChange={(e) => setFormData({ ...formData, subcategory: e.target.value })}
                    placeholder="e.g. Mixer Grinders & Juicers"
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Brand Name</label>
                  <input
                    type="text"
                    value={formData.brand}
                    onChange={(e) => setFormData({ ...formData, brand: e.target.value })}
                    placeholder="e.g. Prestige, LG, Samsung"
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Unit / Pack</label>
                  <input
                    type="text"
                    value={formData.unit}
                    onChange={(e) => setFormData({ ...formData, unit: e.target.value })}
                    placeholder="1 Unit, 5 kg, 500 ml"
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Canteen Price (₹) *</label>
                  <input
                    type="number"
                    required
                    value={formData.price || ''}
                    onChange={(e) =>
                      setFormData({ ...formData, price: Number(e.target.value) })
                    }
                    placeholder="2890"
                    className="w-full mt-1 px-3 py-2 border rounded-lg font-bold text-emerald-950 font-serif"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Original MRP (₹)</label>
                  <input
                    type="number"
                    value={formData.originalPrice || ''}
                    onChange={(e) =>
                      setFormData({ ...formData, originalPrice: Number(e.target.value) })
                    }
                    placeholder="4495"
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>

                <div>
                  <label className="font-bold text-stone-700">Stock Status</label>
                  <select
                    value={formData.stockStatus}
                    onChange={(e) =>
                      setFormData({
                        ...formData,
                        stockStatus: e.target.value as StockStatus,
                      })
                    }
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  >
                    <option value="in_stock">In Stock</option>
                    <option value="low_stock">Low Stock</option>
                    <option value="out_of_stock">Out of Stock</option>
                  </select>
                </div>

                <div>
                  <label className="font-bold text-stone-700">Stock Units Count</label>
                  <input
                    type="number"
                    value={formData.stockCount}
                    onChange={(e) =>
                      setFormData({ ...formData, stockCount: Number(e.target.value) })
                    }
                    className="w-full mt-1 px-3 py-2 border rounded-lg"
                  />
                </div>
              </div>

              <div>
                <label className="font-bold text-stone-700">Image URL</label>
                <input
                  type="url"
                  value={formData.image}
                  onChange={(e) => setFormData({ ...formData, image: e.target.value })}
                  placeholder="https://images.unsplash.com/..."
                  className="w-full mt-1 px-3 py-2 border rounded-lg"
                />
              </div>

              <div>
                <label className="font-bold text-stone-700">Description</label>
                <textarea
                  rows={2}
                  value={formData.description}
                  onChange={(e) => setFormData({ ...formData, description: e.target.value })}
                  placeholder="Product description..."
                  className="w-full mt-1 px-3 py-2 border rounded-lg"
                />
              </div>

              <div>
                <label className="font-bold text-stone-700">
                  Key Features (1 per line)
                </label>
                <textarea
                  rows={2}
                  value={formData.features}
                  onChange={(e) => setFormData({ ...formData, features: e.target.value })}
                  placeholder="Feature 1&#10;Feature 2"
                  className="w-full mt-1 px-3 py-2 border rounded-lg"
                />
              </div>

              <div className="flex items-center gap-4 pt-1">
                <label className="flex items-center gap-2 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={formData.isFeatured}
                    onChange={(e) =>
                      setFormData({ ...formData, isFeatured: e.target.checked })
                    }
                  />
                  <span className="font-bold">Mark as Featured ⭐</span>
                </label>

                <label className="flex items-center gap-2 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={formData.isNewArrival}
                    onChange={(e) =>
                      setFormData({ ...formData, isNewArrival: e.target.checked })
                    }
                  />
                  <span className="font-bold">Mark as New Arrival 🚀</span>
                </label>
              </div>

              <div className="flex justify-end gap-2 pt-3 border-t">
                <button
                  type="button"
                  onClick={() => setIsFormOpen(false)}
                  className="px-4 py-2 border border-stone-300 rounded-lg text-xs font-bold"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-6 py-2 bg-emerald-950 text-amber-300 rounded-lg text-xs font-black shadow hover:bg-emerald-900"
                >
                  Save Product
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
